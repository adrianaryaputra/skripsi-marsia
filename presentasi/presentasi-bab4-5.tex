\begin{frame}{Temuan Lapangan dari Wawancara dan Angket}

        \begin{itemize}

\begin{frame}{Segmen Pelanggan dan Bauran Pemasaran (7P)}
\small
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
        \begin{block}{Siapa pelanggan yang dituju}
        \begin{itemize}
            \item \strong{Segmentasi:} pekerja, mahasiswa, pelanggan sekitar, dan pembeli jam makan siang.
            \item \strong{Target:} orang yang butuh nasi Padang praktis, porsi besar, rasa kuat, rating bagus, dan promo.
            \item \strong{Posisi yang ingin dibangun:} nasi Padang Medan yang praktis, mengenyangkan, rasanya kuat, dan terpercaya di GrabFood.
        \end{itemize}
        \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
        \begin{block}{Bagian 7P yang paling berpengaruh}
        \begin{itemize}
            \item \textit{Product}: rasa lauk dan menu unggulan.
            \item \textit{Price}: harga perlu dibantu paket dan promo yang terukur.
            \item \textit{Promotion}: promo efektif kalau stok dan dapur siap.
            \item \textit{Process}: bagian paling penting adalah packing, antrean, stok, estimasi, dan item terpisah.
            \item \textit{Physical evidence}: foto, rating, ulasan, label, segel, dan info paket.
        \end{itemize}
        \end{block}
    \end{column}
\end{columns}
\begin{center}
\strong{Intinya:} promo akan lebih efektif kalau proses layanan sudah rapi.
\end{center}
\end{frame}

\begin{frame}{Pilihan Strategi dari SWOT}
\scriptsize
\begin{tabularx}{\textwidth}{@{}L{0.23\textwidth}Y Y@{}}
\toprule
\textbf{EFAS/IFAS} & \textbf{Kekuatan (S)} & \textbf{Kelemahan (W)} \\
\midrule
\textbf{Peluang (O)} &
\textbf{SO}\newline
1. Buat paket menu unggulan: dendeng, ayam pop, dan ayam goreng.\newline
2. Perkuat foto, deskripsi, rating, dan posisi menu unggulan.\newline
3. Hubungkan promo dengan stamp digital dan data pelanggan. &
\textbf{WO}\newline
1. Jalankan promo hanya untuk menu yang stok dan dapurnya siap.\newline
2. Perjelas isi paket, kuah, sambal, segel, label, dan stok.\newline
3. Terapkan antrean dan checklist packing. \\
\addlinespace[0.15cm]
\textbf{Ancaman (T)} &
\textbf{ST}\newline
1. Tonjolkan rasa, porsi, dan menu khas.\newline
2. Gunakan rating dan respons keluhan untuk menjaga kepercayaan.\newline
3. Batasi promo agar keuntungan tetap aman. &
\textbf{WT}\newline

\begin{frame}{Kesimpulan}
\small
\begin{enumerate}
    \item Nasi Gerilya punya modal kuat: rasa, porsi, reputasi digital, kualitas bahan, konsistensi rasa, promo, dan kemauan memakai sistem digital.
    \item Masalah utama ada pada proses layanan: packing, item terpisah atau berkuah, antrean, stok, harga di aplikasi, dan info paket.
    \item IFAS 2,50 dan EFAS 2,51 menunjukkan kondisi sedang; posisi SWOT ada di Kuadran I.
    \item Strategi SO menjadi arah utama, tetapi harus dibarengi perbaikan proses agar peluang digital tidak berubah menjadi keluhan.
\end{enumerate}
\vspace{0.15cm}
\begin{block}{Inti kesimpulan}
Penjualan dapat didorong lewat menu unggulan, reputasi digital, promo yang terukur, layanan yang rapi, dan data pelanggan untuk pembelian ulang.
\end{block}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Saran untuk Nasi Gerilya}
\small
\begin{block}{Yang paling perlu dilakukan}
\begin{enumerate}
    \item Pisahkan dan rapikan antrean \textit{dine in}, ojek online, dan \textit{take away}.
    \item Pakai checklist packing untuk nomor pesanan, lauk, kuah, sambal, add-on, catatan khusus, label, dan segel.
    \item Siapkan SOP jam ramai: petugas stok, packing, dan estimasi tunggu driver.
    \item Buat paket menu unggulan yang mudah dipilih, terutama dendeng, ayam pop, dan ayam goreng.
    \item Perjelas foto, isi paket, stok, dan keterangan menu di GrabFood/menu digital.
    \item Jalankan promo hanya saat stok, dapur, biaya/HPP, dan keuntungan aman.
    \item Gunakan stamp digital, data pelanggan, dan Flowie untuk mendorong pembelian ulang.
\end{enumerate}
\end{block}
\vspace{0.1cm}
\begin{center}
\strong{Intinya:} perbaiki proses layanan dulu, lalu perkuat promo dan paket menu.
\end{center}
\end{frame}

\begin{frame}{Keterbatasan Penelitian}
\small
\begin{block}{Hal yang perlu diingat}
\begin{itemize}
    \item Fokus penelitian adalah GrabFood, jadi belum mewakili semua jalur penjualan Nasi Gerilya.
    \item Strategi yang disusun masih berupa rekomendasi, belum diuji langsung menaikkan penjualan.
    \item Jumlah responden pelanggan terbatas, sehingga temuan pelanggan dibaca sebagai gambaran pengalaman.
    \item Wawancara kompetitor belum mencakup semua restoran Padang pesaing di Medan.
    \item Data keuntungan, HPP rinci, dan target BEP tidak dibuka karena termasuk data internal usaha.
    \item Harga, promo, rating, dan ulasan GrabFood bisa berubah.
\end{itemize}
\end{block}
\vspace{0.1cm}
\begin{center}
\strong{Maknanya:} hasil penelitian sudah bisa menjadi arahan strategi, tetapi dampaknya tetap perlu diuji dengan data lanjutan.
\end{center}
\end{frame}
